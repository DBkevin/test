88:        internal fun normalizeCardText(text: String): String {
89:            return text
90:                .replace("[\\u200B-\\u200D\\uFEFF]".toRegex(), "")
91:                .replace("\\s+".toRegex(), " ")

97:        internal fun parseGroupBuyCardText(cardText: String): ParsedCard? {
98:            val normalized = normalizeCardText(cardText)
99:            if (normalized.isBlank()) return null
100:
101:            val title = TITLE_REGEX.find(normalized)?.groupValues?.getOrNull(1)
102:                ?.let(::normalizeCardText)
103:                ?.trim(',', '，')
104:                .orEmpty()
105:            if (title.isBlank()) return null
106:
107:            val displayPrices = DISPLAY_PRICE_REGEX.findAll(normalized)
108:                .map { it.groupValues.getOrNull(1).orEmpty() }
109:                .filter { it.isNotBlank() }
110:                .toList()
111:
112:            val price = CURRENT_PRICE_REGEX.find(normalized)?.groupValues?.getOrNull(1)
113:                ?: displayPrices.getOrNull(0)
114:                ?: ""
115:            val originalPrice = ORIGINAL_PRICE_REGEX.find(normalized)?.groupValues?.getOrNull(1)
116:                ?: displayPrices.getOrNull(1)
117:                ?: ""
118:            val sales = SALES_REGEX.find(normalized)?.groupValues?.getOrNull(1)
119:                ?: BROWSE_REGEX.find(normalized)?.groupValues?.getOrNull(1)
120:                ?: ""
121:
122:            val hasCommerceHint = CARD_HINT_KEYWORDS.any { normalized.contains(it) }
123:            if (price.isBlank() || (!hasCommerceHint && sales.isBlank())) {
124:                return null
125:            }
126:
127:            return ParsedCard(
128:                title = title,
129:                price = price,
130:                originalPrice = originalPrice,
131:                sales = sales,
132:                rawText = normalized
133:            )
134:        }

193:    override fun isTargetPage(nodeInfo: AccessibilityNodeInfo?): Boolean {
194:        if (nodeInfo == null) return false
195:
196:        val pageText = NodeUtils.getAllNodeText(
197:            nodeInfo,
198:            maxDepth = 24,
199:            maxNodes = 520,
200:            maxTextLength = 10000
201:        )
202:        val visibleCards = findVisibleGroupBuyCards(nodeInfo)
203:        if (containsHardNonGroupBuyModule(pageText) && visibleCards.isEmpty()) {
204:            Log.d(TAG, "Detected hard non-groupbuy module, skip target page")
205:            return false
206:        }
207:
208:        val signalCount = SHOP_PAGE_SIGNALS.count { signal ->
209:            pageText.contains(signal, ignoreCase = true)
210:        }
211:        val hasTailSignals = GROUPBUY_TAIL_SIGNALS.any { signal ->
212:            pageText.contains(signal, ignoreCase = true)
213:        }
214:        val merchantName = extractMerchantName(nodeInfo)
215:        val hasConfiguredKeyword = keywords.any { keyword ->
216:            pageText.contains(keyword, ignoreCase = true)
217:        }
218:
219:        val hasMerchantContext =
220:            merchantName.isNotBlank() ||
221:                signalCount >= 2 ||
222:                (signalCount >= 1 && visibleCards.isNotEmpty()) ||
223:                (hasTailSignals && visibleCards.isNotEmpty()) ||
224:                (lastMerchantName.isNotBlank() && !containsHardNonGroupBuyModule(pageText))
225:        val isTarget = visibleCards.isNotEmpty() && (hasMerchantContext || hasConfiguredKeyword)
226:        if (isTarget) {
227:            Log.i(
228:                TAG,
229:                "Target merchant page detected: merchant=${merchantName.ifBlank { lastMerchantName }}, signals=$signalCount, cards=${visibleCards.size}"
230:            )
231:        }
232:        return isTarget

316:    private fun extractMerchantName(rootNode: AccessibilityNodeInfo): String {
317:        val firstCardTop = findVisibleGroupBuyCards(rootNode)
318:            .map { nodeTop(it) }
319:            .minOrNull()
320:            ?: Int.MAX_VALUE
321:
322:        val candidates = NodeUtils.findNodesByCondition(rootNode, { node ->
323:            val label = NodeUtils.getNodeText(node)
324:            if (label.isBlank()) return@findNodesByCondition false
325:            if (label.contains("现价") || label.contains("已售")) return@findNodesByCondition false
326:            if (containsNonMerchantTextMarker(label)) return@findNodesByCondition false
327:
328:            val rect = Rect()
329:            node.getBoundsInScreen(rect)
330:
331:            rect.top in 300..1400 &&
332:                rect.width() >= 400 &&
333:                rect.height() <= 220 &&
334:                rect.bottom <= firstCardTop - 40 &&
335:                HOSPITAL_KEYWORDS.any { keyword -> label.contains(keyword, ignoreCase = true) }
336:        }, maxDepth = 24)
337:
338:        return candidates
339:            .map { node ->
340:                val rect = Rect().also { node.getBoundsInScreen(it) }
341:                rect.top to normalizeCardText(NodeUtils.getNodeText(node))
342:            }
343:            .sortedWith(compareBy<Pair<Int, String>> { it.first }.thenByDescending { it.second.length })
344:            .map { it.second }
345:            .firstOrNull()
346:            .orEmpty()
347:    }

383:    private fun extractGroupBuyList(rootNode: AccessibilityNodeInfo): List<GroupBuyInfo> {
384:        val cards = findVisibleGroupBuyCards(rootNode)
385:        if (cards.isEmpty()) return emptyList()
386:
387:        val skippedSamples = mutableListOf<String>()
388:        val items = cards.mapNotNull { cardNode ->
389:            extractGroupBuyItem(cardNode, skippedSamples)
390:        }.distinctBy { item ->
391:            listOf(item.title, item.price, item.originalPrice).joinToString("|")
392:        }
393:
394:        Log.d(TAG, "Visible group-buy cards: ${items.size}")
395:        if (items.isEmpty() && skippedSamples.isNotEmpty()) {
396:            Log.d(TAG, "Skipped group-buy samples: ${skippedSamples.joinToString(" || ")}")
397:        }
398:        return items
399:    }

401:    private fun findVisibleGroupBuyCards(rootNode: AccessibilityNodeInfo): List<AccessibilityNodeInfo> {
402:        val candidates = NodeUtils.findNodesByCondition(rootNode, { node ->
403:            val label = getNodeText(node)
404:            if (label.isBlank()) return@findNodesByCondition false
405:            if (!isLikelyGroupBuyCardLabel(label)) return@findNodesByCondition false
406:
407:            val rect = Rect()
408:            node.getBoundsInScreen(rect)
409:
410:            rect.top >= MIN_CARD_TOP &&
411:                rect.width() >= MIN_CARD_WIDTH &&
412:                rect.height() >= MIN_CARD_HEIGHT &&
413:                rect.left <= 120 &&
414:                rect.right >= 1180
415:        }, maxDepth = 32)
416:
417:        return candidates
418:            .distinctBy { nodeBoundsKey(it) }
419:            .sortedBy { nodeTop(it) }
420:    }

422:    private fun extractGroupBuyItem(
423:        cardNode: AccessibilityNodeInfo,
424:        skippedSamples: MutableList<String>
425:    ): GroupBuyInfo? {
426:        val ownLabel = normalizeCardText(NodeUtils.getNodeText(cardNode))
427:        val rawCardText = if (ownLabel.isNotBlank()) ownLabel else normalizeCardText(getNodeText(cardNode))
428:        val parsed = parseGroupBuyCardText(rawCardText)
429:        if (parsed == null) {
430:            if (skippedSamples.size < 3 && rawCardText.isNotBlank()) {
431:                skippedSamples.add(rawCardText.take(140))
432:            }
433:            return null
434:        }
435:
436:        return GroupBuyInfo(
437:            title = parsed.title,
438:            price = parsed.price,
439:            originalPrice = parsed.originalPrice,
440:            sales = parsed.sales,
441:            rawText = parsed.rawText
442:        )
443:    }

445:    private fun isLikelyGroupBuyCardLabel(label: String): Boolean {
446:        val normalized = normalizeCardText(label)
447:        if (normalized.isBlank()) return false

449:        val hasPrice = normalized.contains("现价") ||
450:            normalized.contains("原价") ||

470:    private fun getNodeText(node: AccessibilityNodeInfo): String {
471:        val sb = StringBuilder()
472:        collectNodeText(node, sb, 0)
473:        return normalizeCardText(sb.toString())

475:
476:    private fun collectNodeText(node: AccessibilityNodeInfo, sb: StringBuilder, depth: Int) {
477:        if (depth > 12) return
478:
479:        node.text?.let { sb.append(it.toString()).append(" ") }
480:        node.contentDescription?.let { sb.append(it.toString()).append(" ") }
481:
482:        for (i in 0 until node.childCount) {
483:            val child = node.getChild(i)
484:            if (child != null) {
485:                collectNodeText(child, sb, depth + 1)
486:                child.recycle()
487:            }
488:        }
489:    }
