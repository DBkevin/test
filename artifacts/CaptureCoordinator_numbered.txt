77:    suspend fun executeTask(
78:        task: HospitalTask,
79:        launchTargetApp: Boolean = true
80:    ): CaptureExecutionResult = executionMutex.withLock {
81:        val targetPackage = resolveTargetPackage(task)
82:        val appPlugin = service.appPluginManager.findPluginForPackage(targetPackage)
83:
84:        task.targetPackage = targetPackage
85:
86:        val activeExecution = ActiveCapture(
87:            task = task,
88:            targetPackage = targetPackage,
89:            navigationPluginId = appPlugin?.pluginId,
90:            stage = if (launchTargetApp) CaptureStage.OPENING_APP else CaptureStage.WAITING_TARGET_APP
91:        )
92:        activeCapture = activeExecution
93:
94:        return try {
95:            if (!launchTargetApp) {
96:                Log.i(TAG, "Waiting for user to open target app manually: $targetPackage")
97:            }
98:
99:            if (launchTargetApp && !service.launchTargetApp(targetPackage)) {
100:                failure(task, targetPackage, "无法打开目标 App: $targetPackage")
101:            } else if (!waitForPackage(
102:                    targetPackage,
103:                    if (launchTargetApp) 15_000L else resolveManualAppWaitTimeoutMs()
104:                )
105:            ) {
106:                failure(task, targetPackage, "等待目标 App 超时: $targetPackage")
107:            } else {
108:                if (!launchTargetApp) {
109:                    Log.i(TAG, "Target app opened manually, continue capture: $targetPackage")
110:                }
111:                delay(resolveAppStartDelayMs(appPlugin))
112:
113:                val attemptResult = when {
114:                    appPlugin?.captureFlow != null -> executePluginCapture(activeExecution, appPlugin)
115:                    isDouyinPackage(targetPackage) -> executeDouyinCapture(activeExecution)
116:                    else -> executeGenericCapture(activeExecution)
117:                }
118:
119:                if (attemptResult.records.isEmpty()) {
120:                    failure(
121:                        task,
122:                        targetPackage,
123:                        attemptResult.errorMessage ?: "等待抓取结果超时或未产出数据"
124:                    )
125:                } else {
126:                    service.dataStore.saveData(attemptResult.records)
127:                    success(task, targetPackage, attemptResult.records)
128:                }
129:            }
130:        } catch (e: CancellationException) {
131:            Log.i(TAG, "任务被取消，等待外层决定是否恢复: ${task.hospitalName}")
132:            throw e
133:        } catch (e: Exception) {
134:            Log.e(TAG, "任务执行异常: ${task.hospitalName}", e)
135:            failure(task, targetPackage, e.message ?: "未知异常")
136:        } finally {
137:            activeCapture = null
138:        }

419:    private suspend fun collectDetailRecords(
420:        activeExecution: ActiveCapture,
421:        collectionConfig: CollectionConfig?
422:    ): List<ScrapedData> {
423:        val deadline = System.currentTimeMillis() + resolveCaptureTimeoutMs(collectionConfig)
424:        val maxScrollRounds = resolveMaxScrollRounds(collectionConfig)
425:        val maxIdleScrollRounds = resolveMaxIdleScrollRounds(collectionConfig)
426:
427:        var idleRounds = 0
428:        var scrollRounds = 0
429:
430:        val initialViewportReady = stabilizeInitialViewport(
431:            activeExecution = activeExecution,
432:            collectionConfig = collectionConfig,
433:            deadline = deadline
434:        )
435:        if (!initialViewportReady) {
436:            Log.w(TAG, "Initial Douyin viewport not stable enough for scrolling, stop current capture")
437:            return activeExecution.snapshotRecords()
438:        }
439:
440:        if (shouldStopCollection(collectionConfig)) {
441:            Log.i(TAG, "Stop collection after initial viewport capture")
442:            return activeExecution.snapshotRecords()
443:        }
444:
445:        while (scrollRounds < maxScrollRounds && remainingTime(deadline) > 0) {
446:            if (!ensureMerchantDetailContext(activeExecution, "before_scroll:$scrollRounds")) {
447:                Log.i(TAG, "Stop collection after losing merchant detail context before scroll")
448:                break
449:            }
450:
451:            if (shouldStopCollection(collectionConfig)) {
452:                Log.i(TAG, "Stop collection before next scroll: round=$scrollRounds")
453:                break
454:            }
455:
456:            val beforeCount = activeExecution.recordCount()
457:            val beforeRevision = activeExecution.revision()
458:            val scrolled = searchController.scrollCurrentPage()
459:            if (!scrolled) {
460:                break
461:            }
462:
463:            scrollRounds++
464:            delay(resolveScrollSettleMs(collectionConfig))
465:            collectCurrentViewport(activeExecution, collectionConfig, deadline)
466:
467:            if (!ensureMerchantDetailContext(activeExecution, "after_scroll:$scrollRounds")) {
468:                Log.i(TAG, "Stop collection after losing merchant detail context after scroll")
469:                break
470:            }
471:
472:            if (shouldStopCollection(collectionConfig)) {
473:                Log.i(TAG, "Stop collection after scroll round=$scrollRounds")
474:                break
475:            }
476:
477:            val afterCount = activeExecution.recordCount()
478:            val afterRevision = activeExecution.revision()
479:            if (afterCount <= beforeCount && afterRevision <= beforeRevision) {
480:                idleRounds++
481:                if (idleRounds >= maxIdleScrollRounds) {
482:                    break
483:                }
484:            } else {
485:                idleRounds = 0
486:            }
487:        }
488:
489:        return activeExecution.snapshotRecords()
490:    }

492:    private suspend fun stabilizeInitialViewport(
493:        activeExecution: ActiveCapture,
494:        collectionConfig: CollectionConfig?,
495:        deadline: Long
496:    ): Boolean {
497:        repeat(INITIAL_VIEWPORT_MAX_ATTEMPTS) { attempt ->
498:            collectCurrentViewport(activeExecution, collectionConfig, deadline)
499:
500:            if (activeExecution.recordCount() > 0) {
501:                Log.i(TAG, "Initial viewport captured records before first scroll: count=${activeExecution.recordCount()}")
502:                return true
503:            }
504:
505:            val stillOnTargetPage = service.isCurrentTargetPage(activeExecution.targetPackage)
506:            if (stillOnTargetPage) {
507:                Log.i(TAG, "Initial viewport confirmed as target page before first scroll: attempt=${attempt + 1}")
508:                return true
509:            }
510:
511:            val stillOnMerchantDetail = searchController.isOnMerchantDetailPage(
512:                activeExecution.task.hospitalName
513:            )
514:            if (!stillOnMerchantDetail) {
515:                val recovered = searchController.recoverMerchantDetailPage(
516:                    merchantName = activeExecution.task.hospitalName,
517:                    maxBackAttempts = 2
518:                )
519:                if (recovered) {
520:                    Log.i(
521:                        TAG,
522:                        "Recovered merchant detail context during initial viewport stabilization: attempt=${attempt + 1}"
523:                    )
524:                    delay(min(INITIAL_VIEWPORT_RETRY_DELAY_MS, remainingTime(deadline)))
525:                    return@repeat
526:                }
527:                Log.w(
528:                    TAG,
529:                    "Initial viewport lost merchant detail context before first scroll: attempt=${attempt + 1}"
530:                )
531:                return false
532:            }
533:
534:            if (attempt < INITIAL_VIEWPORT_MAX_ATTEMPTS - 1 && remainingTime(deadline) > 0L) {
535:                Log.i(
536:                    TAG,
537:                    "Retry initial viewport capture before first scroll: attempt=${attempt + 1}"
538:                )
539:                delay(min(INITIAL_VIEWPORT_RETRY_DELAY_MS, remainingTime(deadline)))
540:            }
541:        }
542:
543:        return activeExecution.recordCount() > 0 || service.isCurrentTargetPage(activeExecution.targetPackage)
544:    }

546:    private fun shouldStopCollection(collectionConfig: CollectionConfig?): Boolean {
547:        val stopTextsAll = resolveStopTextsAll(collectionConfig)
548:        val stopTextsAny = resolveStopTextsAny(collectionConfig)
549:        val stopTextsNone = resolveStopTextsNone(collectionConfig)
550:        if (stopTextsAll.isEmpty() && stopTextsAny.isEmpty() && stopTextsNone.isEmpty()) {
551:            return false
552:        }
553:
554:        val matchResult = searchController.matchCurrentPageTexts(
555:            requiredAllTexts = stopTextsAll,
556:            requiredAnyTexts = stopTextsAny,
557:            absentTexts = stopTextsNone
558:        )
559:        if (!matchResult.matched) {
560:            return false
561:        }
562:
563:        Log.i(
564:            TAG,
565:            "Detected end-of-list markers: all=${matchResult.matchedAllTexts.joinToString(",")}, any=${matchResult.matchedAnyTexts.joinToString(",")}, none=${stopTextsNone.joinToString(",")}"
566:        )
567:        return true
568:    }

572:        collectionConfig: CollectionConfig?,
573:        deadline: Long
574:    ) {
575:        val baselineRevision = activeExecution.revision()
576:        val expandedCount = expandVisibleSections(collectionConfig)
577:        if (expandedCount > 0) {

584:            activeExecution,

586:            timeoutMs = min(resolveCaptureRoundWaitMs(collectionConfig), remainingTime(deadline))
587:        )

589:        if (!progressed && remainingTime(deadline) > 0L) {
590:            delay(400)
591:            service.scrapeCurrentPageNow(activeExecution.targetPackage)
592:            waitForCaptureProgress(
593:                activeExecution,
594:                baselineRevision = baselineRevision,
595:                timeoutMs = min(900L, remainingTime(deadline))
596:            )
597:        }
598:    }

876:private fun ActiveCapture.appendRecords(records: List<ScrapedData>): CaptureAppendResult {
877:    synchronized(this) {
878:        var addedCount = 0
879:        var updatedCount = 0
880:        records.forEach { record ->
881:            val key = ScrapedRecordIdentity.buildBusinessKey(record)
882:            val existing = recordsByKey[key]
883:            if (existing == null) {
884:                recordsByKey[key] = record
885:                addedCount++
886:            } else {
887:                val merged = ScrapedRecordIdentity.merge(existing, record)
888:                if (merged != existing) {
889:                    recordsByKey[key] = merged
890:                    updatedCount++
891:                }
892:            }
893:        }
894:        if (addedCount > 0 || updatedCount > 0) {
895:            progressRevision++
896:        }
897:        return CaptureAppendResult(addedCount = addedCount, updatedCount = updatedCount)
898:    }
899:}
