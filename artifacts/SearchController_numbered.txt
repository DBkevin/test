119:        keyword: String,
120:        entryKeywords: List<String> = SEARCH_KEYWORDS,
121:        buttonKeywords: List<String> = SEARCH_BUTTON_KEYWORDS
122:    ): Boolean {
123:        val resolvedEntryKeywords = entryKeywords.ifEmpty { SEARCH_KEYWORDS }
124:        val resolvedButtonKeywords = buttonKeywords.ifEmpty { SEARCH_BUTTON_KEYWORDS }
125:        Log.i(TAG, "Searching for: $keyword")
126:        var searchBox: AccessibilityNodeInfo? = null
127:
128:        return try {
129:            // 1. 定位搜索框
130:            searchBox = prepareSearchBox(resolvedEntryKeywords)
131:            if (searchBox == null) {
132:                Log.e(TAG, "Search box not found")
133:                return false
134:            }
135:            
136:            // 2. 清空搜索框
137:            clearSearchBox(searchBox)
138:            
139:            // 3. 输入搜索内容
140:            inputText(searchBox, keyword)
141:
142:            // 4. 点击搜索按钮
143:            val submitted = clickSearchButton(searchBox, resolvedButtonKeywords)
144:            
145:            Log.i(TAG, "Search executed: $keyword, submitted=$submitted")
146:            submitted
147:            
148:        } catch (e: Exception) {

366:    fun clickAnyText(
367:        targetTexts: List<String>,
368:        exactMatch: Boolean = false,
369:        maxClicks: Int = targetTexts.size
370:    ): Int {
371:        val normalizedTargets = targetTexts
372:            .map { it.trim() }
373:            .filter { it.isNotEmpty() }
374:            .distinct()
375:
376:        if (normalizedTargets.isEmpty() || maxClicks <= 0) {
377:            return 0
378:        }
379:
380:        val clickedKeys = mutableSetOf<String>()
381:        var clickCount = 0
382:
383:        while (clickCount < maxClicks) {
384:            var clickedInThisPass = false
385:
386:            for (targetText in normalizedTargets) {
387:                val matchedNode = findTextNode(targetText, exactMatch) ?: continue
388:                try {
389:                    val bounds = Rect()
390:                    matchedNode.getBoundsInScreen(bounds)
391:                    val clickKey = "$targetText:${bounds.left},${bounds.top},${bounds.right},${bounds.bottom}"
392:                    if (!clickedKeys.add(clickKey)) {
393:                        continue
394:                    }
395:
396:                    val clicked = clickNodeWithTrace("click_any_text:$targetText", matchedNode)
397:                    Log.i(
398:                        TAG,
399:                        "Click any text: target=$targetText, exact=$exactMatch, clicked=$clicked"
400:                    )
401:                    if (clicked) {
402:                        clickCount++
403:                        clickedInThisPass = true
404:                        Thread.sleep(600)
405:                        break
406:                    }
407:                } finally {
408:                    matchedNode.recycle()
409:                }
410:            }
411:
412:            if (!clickedInThisPass) {
413:                break
414:            }
415:        }
416:
417:        return clickCount
418:    }

420:    fun clickMerchantExpandTexts(
421:        targetTexts: List<String>,
422:        maxClicks: Int = targetTexts.size
423:    ): Int {
424:        val normalizedTargets = targetTexts
425:            .map { normalizeText(it) }
426:            .filter { it.isNotBlank() }
427:            .distinct()
428:        if (normalizedTargets.isEmpty() || maxClicks <= 0) {
429:            return 0
430:        }
431:
432:        val rootNode = service.rootInActiveWindow ?: return 0
433:        val metrics = service.resources.displayMetrics
434:        val minTop = (metrics.heightPixels * 0.48f).toInt()
435:        val maxBottom = (metrics.heightPixels * 0.93f).toInt()
436:
437:        try {
438:            val candidates = NodeUtils.findNodesByCondition(
439:                rootNode,
440:                condition = { node: AccessibilityNodeInfo ->
441:                    val rawText = getComparableNodeText(node)
442:                    if (rawText.isBlank()) {
443:                        return@findNodesByCondition false
444:                    }
445:                    if (rawText.contains("你可能感兴趣", ignoreCase = true) ||
446:                        rawText.contains("猜你喜欢", ignoreCase = true)
447:                    ) {
448:                        return@findNodesByCondition false
449:                    }
450:
451:                    val normalizedText = normalizeText(rawText)
452:                    val keywordMatched = normalizedTargets.any { target ->
453:                        normalizedText.contains(target)
454:                    }
455:                    if (!keywordMatched) {
456:                        return@findNodesByCondition false
457:                    }
458:
459:                    val bounds = Rect().also { node.getBoundsInScreen(it) }
460:                    !bounds.isEmpty &&
461:                        bounds.top >= minTop &&
462:                        bounds.bottom <= maxBottom &&
463:                        bounds.height() in 36..220
464:                },
465:                maxDepth = 28
466:            )
467:
468:            try {
469:                val sortedCandidates = candidates.sortedWith(
470:                    compareBy<AccessibilityNodeInfo>({ node ->
471:                        Rect().also { node.getBoundsInScreen(it) }.top
472:                    }, { node ->
473:                        Rect().also { node.getBoundsInScreen(it) }.left
474:                    })
475:                )
476:
477:                val clickedKeys = mutableSetOf<String>()
478:                var clickCount = 0
479:
480:                for (candidate in sortedCandidates) {
481:                    if (clickCount >= maxClicks) {
482:                        break
483:                    }
484:
485:                    val bounds = Rect()
486:                    candidate.getBoundsInScreen(bounds)
487:                    if (bounds.isEmpty) {
488:                        continue
489:                    }
490:
491:                    val clickKey = "${bounds.left},${bounds.top},${bounds.right},${bounds.bottom}"
492:                    if (!clickedKeys.add(clickKey)) {
493:                        continue
494:                    }
495:
496:                    val clicked = clickNodeWithTrace("merchant_expand", candidate)
497:                    if (clicked) {
498:                        clickCount++
499:                        Thread.sleep(500)
500:                    }
501:                }
502:
503:                if (clickCount > 0) {
504:                    Log.i(TAG, "Merchant expand taps: count=$clickCount")
505:                }
506:                return clickCount
507:            } finally {
508:                NodeUtils.recycleNodes(candidates)
509:            }
510:        } finally {
511:            rootNode.recycle()
512:        }
513:    }

624:        return try {

635:    fun matchCurrentPageTexts(
636:        requiredAllTexts: List<String> = emptyList(),
637:        requiredAnyTexts: List<String> = emptyList(),
638:        absentTexts: List<String> = emptyList()
639:    ): PageTextMatchResult {
640:        val normalizedAllTexts = requiredAllTexts
641:            .map { it.trim() }
642:            .filter { it.isNotEmpty() }
643:            .distinct()
644:        val normalizedAnyTexts = requiredAnyTexts
645:            .map { it.trim() }
646:            .filter { it.isNotEmpty() }
647:            .distinct()
648:        val normalizedAbsentTexts = absentTexts
649:            .map { it.trim() }
650:            .filter { it.isNotEmpty() }
651:            .distinct()
652:
653:        if (normalizedAllTexts.isEmpty() && normalizedAnyTexts.isEmpty() && normalizedAbsentTexts.isEmpty()) {
654:            return PageTextMatchResult(matched = false)
655:        }
656:
657:        val rootNode = service.rootInActiveWindow ?: return PageTextMatchResult(matched = false)
658:
659:        return try {
660:            val pageText = NodeUtils.getAllNodeText(
661:                rootNode,
662:                maxDepth = 18,
663:                maxNodes = 420,
664:                maxTextLength = 7000
665:            )
666:            val matchedAllTexts = normalizedAllTexts.filter { target ->
667:                pageText.contains(target, ignoreCase = true)
668:            }
669:            val matchedAnyTexts = normalizedAnyTexts.filter { target ->
670:                pageText.contains(target, ignoreCase = true)
671:            }
672:            val presentNoneTexts = normalizedAbsentTexts.filter { target ->
673:                pageText.contains(target, ignoreCase = true)
674:            }
675:            val allMatched = normalizedAllTexts.isEmpty() ||
676:                matchedAllTexts.size == normalizedAllTexts.size
677:            val anyMatched = normalizedAnyTexts.isEmpty() || matchedAnyTexts.isNotEmpty()
678:            val noneMatched = presentNoneTexts.isEmpty()
679:
680:            PageTextMatchResult(
681:                matched = allMatched && anyMatched && noneMatched,
682:                matchedAllTexts = matchedAllTexts,
683:                matchedAnyTexts = matchedAnyTexts,
684:                presentNoneTexts = presentNoneTexts
685:            )
686:        } finally {
687:            rootNode.recycle()
688:        }
689:    }

710:        val rootNode = service.rootInActiveWindow ?: return null
711:        
712:        try {
713:            // 方法 1: 通过更稳定的 viewId 查找真实输入框
714:            val byId = NodeUtils.findNodeByCondition(
715:                rootNode,
716:                condition = { node: AccessibilityNodeInfo ->
717:                    val viewId = node.viewIdResourceName?.lowercase() ?: return@findNodeByCondition false

720:                        viewId.contains("search_input") ||
721:                        viewId.contains("search_edit")) &&

724:                maxDepth = SEARCH_NODE_MAX_DEPTH

727:            if (byId != null) {
728:                Log.d(TAG, "Found search box by id")
729:                return AccessibilityNodeInfo.obtain(byId)
730:            }
731:
732:            // 方法 2: 通过 className / editability 查找真实输入框
733:            val byClass = NodeUtils.findNodeByCondition(rootNode, condition = { node: AccessibilityNodeInfo ->
734:                isLikelySearchInput(node)
735:            }, maxDepth = SEARCH_NODE_MAX_DEPTH)
736:

738:                Log.d(TAG, "Found search box by class")
739:                return AccessibilityNodeInfo.obtain(byClass)
740:            }
741:            
742:            // 方法 3: 通过关键词查找可输入的搜索框，避免把顶部搜索按钮误当成输入框
743:            val byKeyword = NodeUtils.findNodeByCondition(
744:                rootNode,
745:                condition = { node: AccessibilityNodeInfo ->
746:                    if (!isLikelySearchInput(node)) {
747:                        return@findNodeByCondition false

751:                    searchKeywords.any { keyword -> text.contains(keyword.lowercase()) }
752:                },
753:                maxDepth = SEARCH_NODE_MAX_DEPTH
754:            )
755:
756:            if (byKeyword != null) {
757:                Log.d(TAG, "Found search box by keyword")
758:                return AccessibilityNodeInfo.obtain(byKeyword)
759:            }
760:            
761:            return null
762:            
763:        } finally {
764:            rootNode.recycle()
765:        }
