69:        internal fun resolveKind(signals: DouyinPageSignals): DouyinPageKind {
70:            if (signals.hasSearchInput && signals.hasSearchSubmitButton) {
71:                return DouyinPageKind.GROUPBUY_SEARCH_INPUT
72:            }
73:            if (signals.hasMerchantHeaderAnchor && signals.hasMerchantBottomActionBar) {
74:                return DouyinPageKind.MERCHANT_HOME
75:            }
76:            if (signals.hasMerchantBottomActionBar &&
77:                signals.hasMerchantTailSignal &&
78:                signals.hasMerchantCommerceSignal
79:            ) {
80:                return DouyinPageKind.MERCHANT_TAIL
81:            }
82:            if (signals.hasRecommendationSignal) {
83:                return DouyinPageKind.RECOMMENDATION
84:            }
85:            if (signals.hasMerchantNodeForTarget && signals.hasMerchantResultSignals) {
86:                return DouyinPageKind.MERCHANT_RESULT_LIST
87:            }
88:            if ((signals.hasSelectedGroupBuyTab || signals.hasSearchEntryNode) &&
89:                (signals.hasTopSearchButton || signals.hasSearchSignal || signals.hasSearchEntryNode) &&
90:                (signals.hasLocationSignal || signals.hasBottomHomeTab || signals.hasGroupBuyKeywordCluster)
91:            ) {
92:                return DouyinPageKind.GROUPBUY_HOME
93:            }
94:            if (signals.hasBottomHomeTab && !signals.hasSelectedGroupBuyTab) {
95:                return DouyinPageKind.HOME_FEED
96:            }
97:            return DouyinPageKind.UNKNOWN
98:        }

271:    private fun hasMerchantHeaderAnchor(
272:        rootNode: AccessibilityNodeInfo,
273:        normalizedTarget: String
274:    ): Boolean {
275:        val nodes = NodeUtils.findNodesByCondition(
276:            rootNode,
277:            condition = { node ->
278:                val text = NodeUtils.getNodeText(node)
279:                if (text.isBlank()) {
280:                    false
281:                } else {
282:                    val bounds = Rect().also { node.getBoundsInScreen(it) }
283:                    !bounds.isEmpty &&
284:                        bounds.top in 450..1500 &&
285:                        bounds.bottom <= 1700
286:                }
287:            },
288:            maxDepth = 24
289:        )
290:
291:        var hasMerchantTitle = normalizedTarget.isBlank()
292:        var hintMatches = 0
293:
294:        try {
295:            nodes.forEach { node ->
296:                val text = NodeUtils.getNodeText(node)
297:                val normalizedText = normalizeText(text)
298:                if (!hasMerchantTitle &&
299:                    normalizedTarget.isNotBlank() &&
300:                    normalizedText.contains(normalizedTarget)
301:                ) {
302:                    hasMerchantTitle = true
303:                }
304:                if (MERCHANT_HOME_TOP_HINTS.any { hint -> text.contains(hint, ignoreCase = true) }) {
305:                    hintMatches++
306:                }
307:            }
308:        } finally {
309:            NodeUtils.recycleNodes(nodes)
310:        }
311:
312:        return hasMerchantTitle && hintMatches > 0
313:    }

315:    private fun hasMerchantBottomActionBarAnchor(
316:        rootNode: AccessibilityNodeInfo,
317:        pageText: String
318:    ): Boolean {
319:        val metrics = service.resources.displayMetrics
320:        val minTop = (metrics.heightPixels * 0.88f).toInt()
321:        val minWidth = (metrics.widthPixels * 0.98f).toInt()
322:        val hasBottomTexts = BOTTOM_ACTION_BAR_HINTS.all { hint ->
323:            pageText.contains(hint, ignoreCase = true)
324:        }
325:        if (!hasBottomTexts) {
326:            return false
327:        }
328:
329:        val bottomBarNode = NodeUtils.findNodeByCondition(
330:            rootNode,
331:            condition = { node ->
332:                val bounds = Rect().also { node.getBoundsInScreen(it) }
333:                !bounds.isEmpty &&
334:                    bounds.top >= minTop &&
335:                    bounds.left <= 10 &&
336:                    bounds.right >= metrics.widthPixels - 10 &&
337:                    bounds.width() >= minWidth &&
338:                    bounds.height() in 140..260 &&
339:                    !node.isScrollable
340:            },
341:            maxDepth = 18
342:        )
343:
344:        val matched = bottomBarNode != null
345:        bottomBarNode?.recycle()
346:        return matched
347:    }

349:    private fun isLikelySearchInput(node: AccessibilityNodeInfo): Boolean {
350:        if (!node.isVisibleToUser) return false
351:
352:        val className = node.className?.toString().orEmpty()
353:        val viewId = node.viewIdResourceName?.lowercase().orEmpty()
354:        val text = NodeUtils.getNodeText(node)
355:        val bounds = Rect().also { node.getBoundsInScreen(it) }
356:        val looksLikeEditableClass =
357:            className.contains("EditText", ignoreCase = true) ||
358:                className.contains("AutoCompleteTextView", ignoreCase = true)
359:        val looksLikeSearchFieldId =
360:            viewId.contains("search") ||
361:                viewId.contains("et_search") ||
362:                viewId.contains("search_kw") ||
363:                viewId.contains("search_input")
364:        val hintLikeSearch = text.contains("搜索", ignoreCase = true) ||
365:            node.hintText?.toString()?.contains("搜索", ignoreCase = true) == true
366:
367:        return (looksLikeEditableClass || looksLikeSearchFieldId || node.isEditable) &&
368:            bounds.width() >= 300 &&
369:            bounds.height() >= 80 &&
370:            (hintLikeSearch || looksLikeSearchFieldId || looksLikeEditableClass)
371:    }
